from openai import OpenAI

client = OpenAI(
    api_key=os.environ["TRUSTEDROUTER_API_KEY"],
    base_url="{{ api_base_url }}",
)

response = client.chat.completions.create(
    model="trustedrouter/zdr",
    messages=[{
        "role": "user",
        "content": "Reply PONG",
    }],
)

print(response.choices[0].message.content)
